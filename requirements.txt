# Extra deps the fleet ComfyUI needs for the speech-to-speech custom nodes,
# on top of a stock ComfyUI + OmniVoice install. Install into the ComfyUI venv:
#   pip install -r requirements.txt
#
# audio-separator provides the Mel-Band Roformer vocal separation used by
# audio_extract_vocals.py (far cleaner instrumental than HDemucs — no ghost
# voice left in the backing track). The Roformer model
# (model_mel_band_roformer_ep_3005_sdr_11.4360.ckpt) auto-downloads on first
# run into the audio-separator model dir.
#
# audio-separator >=0.46: audio_extract_vocals.py passes use_native_fp16,
# which older releases reject. Do NOT install seed-vc into the same venv —
# it pins numpy<2 and pip would silently downgrade audio-separator to satisfy
# it (see requirements-seedvc.txt).
#
# torchaudio: imported by audio_overlay_voice.py (and by omnivoice). Nothing
# else pulls it in; its final release (2.11.0) is forward-compatible with
# newer torch, so it is left unpinned.
audio-separator[gpu]>=0.46
